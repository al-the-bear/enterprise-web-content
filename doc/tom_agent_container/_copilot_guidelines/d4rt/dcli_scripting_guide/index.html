DCli Scripting Guide

Complete reference for shell scripting in Dart using the DCli library.

---

Execution Environments

This guide covers scripting with DCli in three execution contexts:

EnvironmentCommandDescription
Compiled dart compile exe script.dart Native executable, no runtime required
JIT dart run script.dart Standard Dart execution with JIT compilation
dcli dcli script.dart or ./script.dart DCli tool with shebang support
tom tom script.dart Tom CLI with full workspace integration

Shebang Usage

Add a shebang line to make scripts directly executable:

#!/usr/bin/env dcli
// or: #!/usr/bin/env dart
// or: #!/usr/bin/env d4rt

import 'package:dcli/dcli.dart';

void main() {
  print('Hello from DCli!');
}

Then run: chmod +x script.dart && ./script.dart

Getting Started

Installation

Add DCli to your pubspec.yaml:

dependencies:
  dcli: ^8.4.2

Basic Import

import 'package:dcli/dcli.dart';

This single import provides access to all DCli functions, classes, and extensions.

StringAsProcess Extension

Key Classes

Progress

Captures command output:

var progress = start('ls -la', progress: Progress.print());

// Or capture to list
var progress = start('git log --oneline -5');
for (var line in progress.lines) {
  print('Line: $line');
}

Static constructors: - Progress.print() — Print to stdout - Progress.devNull() — Discard output - Progress.capture() — Capture to .lines

Env

Environment variable access:

var home = env['HOME'];
env['MY_VAR'] = 'value';
env.appendToPATH('/custom/bin');

DartSdk

Access to Dart SDK:

var sdk = DartSdk();
print(sdk.pathToSdk);
print(sdk.version);
sdk.dart.runScript('script.dart');

Shell

Shell detection and utilities:

var shell = Shell.current;
print(shell.name);
print(shell.loggedInUser);

NamedLock

Cross-process file locking:

NamedLock(name: 'mylock').withLock(() {
  // Protected code
});

Find / FindItem

File searching with detailed results:

find('*.dart', recursive: true).forEach((item) {
  print('${item.pathTo} - ${item.length} bytes');
});

Examples

Basic File Operations

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main() {
  // Navigate and list
  print('Current directory: $pwd');
  print('Home: $HOME');
  
  // Create directory structure
  var workDir = createDir(join(HOME, 'dcli_demo'), recursive: true);
  
  // Create and write file
  var filePath = join(workDir, 'hello.txt');
  filePath.write('Hello, DCli!');
  filePath.append('This is line 2');
  
  // Read and display
  print('\nFile contents:');
  cat(filePath);
  
  // Copy and move
  var copyPath = join(workDir, 'hello_copy.txt');
  copy(filePath, copyPath);
  
  // Find files
  print('\nText files in demo:');
  find('*.txt', workingDirectory: workDir).forEach(print);
  
  // Cleanup
  deleteDir(workDir);
}

Command Execution

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main() {
  // Simple run
  'echo "Hello from shell"'.run;
  
  // Capture output
  var dartVersion = 'dart --version'.firstLine;
  print('Dart: $dartVersion');
  
  // Process lines
  'git log --oneline -5'.forEach((line) {
    print('  Commit: $line');
  });
  
  // Handle errors
  try {
    'false'.run;  // Will throw
  } on RunException catch (e) {
    print('Command failed with exit code: ${e.exitCode}');
  }
  
  // Ignore errors
  'might-not-exist'.toList(nothrow: true);
}

Interactive Input

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main() {
  // Text input with validation
  var name = ask('Enter your name:', 
    required: true, 
    validator: Ask.alpha);
    
  // Numeric input
  var age = ask('Enter your age:', 
    validator: Ask.all([
      Ask.integer,
      Ask.valueRange(1, 150),
    ]));
    
  // Password (hidden)
  var password = ask('Password:', hidden: true);
  
  // Confirmation
  if (confirm('Save profile?', defaultValue: true)) {
    print(green('Profile saved for $name'));
  }
  
  // Menu selection
  var choice = menu('Select option:', options: [
    'View profile',
    'Edit profile', 
    'Delete profile',
  ]);
  print('Selected: $choice');
}

Working with Temp Files

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main() {
  // Temp dir with auto-cleanup
  withTempDir((tmpDir) {
    var dataFile = join(tmpDir, 'data.json');
    dataFile.write('{"key": "value"}');
    
    // Process file...
    print('Working in: $tmpDir');
    
    // Auto-deleted when scope exits
  });
  
  // Keep temp dir for debugging
  var debugDir = withTempDir((tmp) {
    print('Debug dir: $tmp');
    return tmp;
  }, keep: true);
}

Downloading Files

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main() async {
  var url = 'https://example.com/file.zip';
  var savePath = join(HOME, 'Downloads', 'file.zip');
  
  await fetch(
    url: url,
    saveToPath: savePath,
    fetchProgress: (progress) {
      print('${progress.progress}% downloaded');
    },
  );
  
  print(green('Downloaded to: $savePath'));
}

Build Script Example

#!/usr/bin/env dcli
import 'package:dcli/dcli.dart';

void main(List<String> args) {
  var target = args.isEmpty ? 'all' : args.first;
  
  print(cyan('Building: $target'));
  
  switch (target) {
    case 'clean':
      if (exists('build')) deleteDir('build');
      print(green('Cleaned'));
      
    case 'compile':
      createDir('build', recursive: true);
      'dart compile exe bin/main.dart -o build/app'.run;
      print(green('Compiled'));
      
    case 'test':
      'dart test'.run;
      
    case 'all':
      main(['clean']);
      main(['compile']);
      main(['test']);
      
    default:
      printerr(red('Unknown target: $target'));
  }
}

'cmd'.runConsoleYesvoid

Running in Shell

Use runInShell: true when you need: - Shell built-ins (cd, export, etc.) - Pipes and redirects (|, >, <) - Glob expansion by shell - Shell environment features 

// Won't work without shell
'echo $HOME'.run;  // Prints literal $HOME

// Use shell for variable expansion
run('echo \$HOME', runInShell: true);

Background Processes

// Detached process (survives script exit)
'long-running-server'.start(detached: true);

// Background with output capture
var progress = 'tail -f log.txt'.start(
  progress: Progress.capture(),
);

Appendix B: Find Function Details

Find with Type Filters

import 'dart:io';

// Find only files
find('*', types: [FileSystemEntityType.file]).forEach(print);

// Find only directories  
find('*', types: [FileSystemEntityType.directory]).forEach(print);

// Find only links
find('*', types: [FileSystemEntityType.link]).forEach(print);

Advanced Find Patterns

// Case-insensitive search
find('*.PDF', caseSensitive: false);

// Include hidden files
find('.*', includeHidden: true);

// Non-recursive (current dir only)
find('*.dart', recursive: false);

// Custom working directory
find('*.yaml', workingDirectory: '/project');

Appendix C: BuildKit Integration

DCli scripts integrate with BuildKit pipelines for workspace automation.

Pipeline Configuration

In buildkit.yaml:

pipelines:
  deploy:
    steps:
      - dcli scripts/deploy.dart
      - dcli scripts/notify.dart --env production

Running DCli from BuildKit

# Run dcli script via buildkit
bk :dcli scripts/myscript.dart

# With arguments
bk :dcli scripts/myscript.dart -- --verbose --target prod

Inline DCli in Pipelines

pipelines:
  quick-check:
    steps:
      - dcli "print('Hello from inline dcli')"
      - dcli "if (exists('build')) print('Build exists');"

Script Dependencies

For scripts with dependencies, create a proper pubspec:

project/
├── pubspec.yaml
├── scripts/
│   ├── build.dart
│   └── deploy.dart

pubspec.yaml:

name: project_scripts
dependencies:
  dcli: ^8.4.2

just compile with the normal dart compile or configure buildkit for the project and run

buildkit --project <your-project> :compiler

Appendix D: Error Handling

Common Exceptions

ExceptionThrown ByDescription
RunException run, start, .run, .toList() Command failed
CopyException copy, copyTree Copy operation failed
MoveExceptionmoveMove operation failed
DeleteExceptiondeleteDelete operation failed
CreateDirException createDir Directory creation failed
ReadExceptionreadFile read failed
FetchExceptionfetchDownload failed
FileNotFoundExceptionVariousFile doesn't exist

Error Handling Pattern

try {
  copy('source.txt', 'dest.txt');
} on CopyException catch (e) {
  printerr(red('Copy failed: ${e.message}'));
  printerr('  From: ${e.from}');
  printerr('  To: ${e.to}');
  exit(1);
}

Non-Throwing Alternatives

// Check before operating
if (exists('file.txt')) {
  delete('file.txt');
}

// Use nothrow for commands
var result = 'risky-command'.toList(nothrow: true);

// Catch and continue
try {
  'optional-step'.run;
} catch (_) {
  print(yellow('Optional step skipped'));
}

Appendix E: Cross-Platform Considerations

Path Handling

// Always use truepath for absolute paths
var configPath = truepath(HOME, '.config', 'myapp');

// Use join for path construction
var logFile = join(configPath, 'app.log');

// Check platform
if (Platform.isWindows) {
  // Windows-specific handling
}

Line Endings

// Use eol for platform-appropriate line ending
file.write('line 1${eol}line 2');

// Or let DCli handle it
file.write('line 1');  // Adds platform eol
file.append('line 2'); // Adds platform eol

Executable Search

// Find executable on PATH (handles .exe on Windows)
var dart = which('dart').path;
if (dart != null) {
  print('Dart found at: $dart');
}
